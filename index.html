// ========================================== // 1. CONFIGURACIÓN E INICIALIZACIÓN SUPABASE // ========================================== const SUPABASE_URL = 'https://bgrhwfutbmoktvdziamk.supabase.co'; // Reemplazar este string con la clave public anon copiada de Supabase (Project Settings > API) const SUPABASE_ANON_KEY = 'TU_ANON_PUBLIC_KEY_AQUI'; const _supabase = supabase.createClient(SUPABASE_URL, SUPABASE_ANON_KEY); // ========================================== // 2. MÓDULO DE PERSISTENCIA (SUPABASE API) // ========================================== /** * Obtiene la lista completa de alumnos desde la base de datos. */ async function obtenerAlumnosDB() { try { const { data, error } = await _supabase .from('Alumnos') .select('*') .order('nombre', { ascending: true }); if (error) throw error; return data; } catch (err) { console.error('Error al obtener la lista de alumnos:', err.message); return []; } } /** * Registra un nuevo alumno en la base de datos. * Incluye número celular y la fecha límite de pago. */ async function guardarAlumnoDB(nuevoAlumno) { /* Estructura esperada de nuevoAlumno: { num_ficha: '001', nombre: 'Nico Pérez', pass: '1234', celular: '+5491112345678', plan: '2x semana', arancel: 25000, descuento: false, fecha_pago: '2026-09-30', creditos_recupero: 0, turnos_fijos: [{ dia: 'Lunes', hora: '18:00' }, { dia: 'Miércoles', hora: '18:00' }] } */ try { const { data, error } = await _supabase .from('Alumnos') .insert([nuevoAlumno]) .select(); if (error) throw error; console.log('Alumno guardado exitosamente:', data); return data; } catch (err) { console.error('Error al dar de alta el alumno:', err.message); return null; } } /** * Actualiza los datos de un alumno existente por su ID. */ async function actualizarAlumnoDB(id, datosActualizados) { try { const { data, error } = await _supabase .from('Alumnos') .update(datosActualizados) .eq('id', id) .select(); if (error) throw error; return data; } catch (err) { console.error('Error al actualizar el alumno:', err.message); return null; } } /** * Elimina la ficha de un alumno por ID. */ async function eliminarAlumnoDB(id) { try { const { error } = await _supabase .from('Alumnos') .delete() .eq('id', id); if (error) throw error; return true; } catch (err) { console.error('Error al eliminar el alumno:', err.message); return false; } } // ========================================== // 3. LÓGICA DE NEGOCIO Y ESTADO DINÁMICO // ========================================== /** * Evalúa si el alumno está ACTIVO o PENDIENTE en función de la semana visualizada. * * @param {string} fechaPagoAlumno - Fecha límite en formato 'YYYY-MM-DD' (ej: '2026-09-30') * @param {Date|string} fechaInicioSemanaNavegada - Fecha correspondiente al lunes de la semana que se está mirando en la app. * @returns {string} 'ACTIVO' | 'PENDIENTE' */ function calcularEstadoPorSemana(fechaPagoAlumno, fechaInicioSemanaNavegada) { if (!fechaPagoAlumno) return 'PENDIENTE'; // Parseo seguro de fechas sin desfase horario const [pAño, pMes, pDia] = fechaPagoAlumno.split('-').map(Number); const limitePago = new Date(pAño, pMes - 1, pDia, 23, 59, 59); const semanaNavegada = new Date(fechaInicioSemanaNavegada); semanaNavegada.setHours(0, 0, 0, 0); // Si el inicio de la semana que se consulta supera la fecha tope pagada, figura PENDIENTE if (semanaNavegada > limitePago) { return 'PENDIENTE'; } return 'ACTIVO'; } /** * Procesa el objeto del alumno agregando la propiedad calculada 'estado_semanal' * para ser renderizada en la interfaz. */ function obtenerFichaAlumnoParaSemana(alumno, fechaInicioSemanaNavegada) { const estadoDinamico = calcularEstadoPorSemana( alumno.fecha_pago, fechaInicioSemanaNavegada ); return { ...alumno, estado_semanal: estadoDinamico }; }
